--SQL Functions
CREATE FUNCTION AddThreeNumbers (@num1 INT, @num2 INT, @num3 INT)
RETURNS INT
AS
BEGIN
    DECLARE @sum INT;
    SET @sum = @num1 + @num2 + @num3;
    RETURN @sum;
END;

SELECT dbo.AddThreeNumbers(10, 20, 30) AS TotalSum;
